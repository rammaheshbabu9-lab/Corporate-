/* CREATE PNG CANVASES */

async function createImages(){

const front=
await html2canvas(
document.getElementById("frontCard"),
{
scale:2,
useCORS:true,
backgroundColor:"#ffffff"
}
);


const back=
await html2canvas(
document.getElementById("backCard"),
{
scale:2,
useCORS:true,
backgroundColor:"#ffffff"
}
);


return{
front,
back
};

}


/* PNG DOWNLOAD */

async function downloadPNG(){

try{

showToast("Preparing PNG...");

const{front,back}=
await createImages();

const gap=40;

const canvas=
document.createElement("canvas");

canvas.width=
front.width+
gap+
back.width;

canvas.height=
Math.max(front.height,back.height);

const ctx=
canvas.getContext("2d");

ctx.fillStyle="#ffffff";

ctx.fillRect(
0,
0,
canvas.width,
canvas.height
);

ctx.drawImage(front,0,0);

ctx.drawImage(
back,
front.width+gap,
0
);

canvas.toBlob(function(blob){

if(!blob){

alert("PNG creation failed.");

return;
}

const url=
URL.createObjectURL(blob);

const a=
document.createElement("a");

a.href=url;

a.download=
"India-Digital-Profile-"+profile.id+".png";

document.body.appendChild(a);

a.click();

document.body.removeChild(a);

setTimeout(function(){

URL.revokeObjectURL(url);

},1500);

showToast("PNG download started");

},"image/png");

}catch(error){

console.error(error);

alert(
"PNG download failed. Please try again."
);

}

}


/* PDF DOWNLOAD */

async function downloadPDF(){

try{

showToast("Preparing PDF...");

const{front,back}=
await createImages();

const{jsPDF}=
window.jspdf;

const pdf=
new jsPDF({

orientation:"landscape",
unit:"mm",
format:"a4"

});

const pageWidth=
pdf.internal.pageSize.getWidth();

const pageHeight=
pdf.internal.pageSize.getHeight();

const margin=10;
const gap=6;

const cardWidth=
(pageWidth-margin*2-gap)/2;

const cardHeight=
cardWidth*
(front.height/front.width);

const y=
(pageHeight-cardHeight)/2;


pdf.addImage(
front.toDataURL("image/png"),
"PNG",
margin,
y,
cardWidth,
cardHeight
);


pdf.addImage(
back.toDataURL("image/png"),
"PNG",
margin+cardWidth+gap,
y,
cardWidth,
cardHeight
);


pdf.save(
"India-Digital-Profile-"+profile.id+".pdf"
);

showToast("PDF download started");

}catch(error){

console.error(error);

alert(
"PDF download failed. Please try again."
);

}

}

toast.style.display="none";

},2200);

}
